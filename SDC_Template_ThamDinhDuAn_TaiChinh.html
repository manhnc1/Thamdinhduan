Function FD(mycell)
    Dim a, b, F
    If mycell = "" Then
        FD = ""
    Else
        If Left(mycell.Formula, 1) <> "=" Then
            a = mycell.Address(0, 0)
            FD = a + "=Value"
        Else
            a = mycell.Address(0, 0)
            F = mycell.Formula
            FD = a + F
        End If
    End If
End Function


'Tinh thu nhap chiu thue vao mot nam xac dinh
' rg        : Dong tien can tinh Thu nhap chiu thue
' nn        : Nam tinh thue thu nhap chiu thue
' sntllk    : So nam tich luy luy ke

Public Function TNCT(rg As Range, nn As Long, sntllk As Long) As Double

Application.Volatile
Dim i As Long, j As Long, k As Long, n As Long, nho As Double, crg As Long
Dim tn() As Double, tam() As Double

On Error Resume Next

crg = rg.Count

ReDim tn(1 To crg)
ReDim tam(1 To crg)

For i = 1 To crg
    tam(i) = rg.Cells(1, i).Value ' Nap dong thu nhap vao
Next i

For n = 1 To crg
If tam(n) > 0 Then
    tn(n) = tam(n)      'Gan cac thu nhap duong vao mang tn
Else
   tn(n) = 0
   k = n + sntllk
   nho = tam(n)         'Gan TN vao bien nho
   If k <= crg Then
        For j = n + 1 To k
            If tam(j) > 0 Then
                nho = nho + tam(j)
                If nho < 0 Then
                    tam(j) = 0
                Else
                    tam(j) = nho
                    Exit For
                End If
            End If
        Next j
   Else
        For j = n + 1 To crg
            If tam(j) > 0 Then
                nho = nho + tam(j)
                If nho < 0 Then
                    tam(j) = 0
                Else
                    tam(j) = nho
                    Exit For
                End If
            End If
        Next j
   End If
   
End If
Next n

TNCT = tn(nn)

End Function



'Tinh thue thu nhap voi thue suat thay doi theo 3 thoi doan
'Tinh thue thu nhap ke tu khi Thu nhap chiu thue > 0
'n: nam dau co thue suat a%
'm: nam tiep theo co thue sua b%
'sau do co thue suat c%

Public Function ThueTN(Cashflow As Range, nTax As Long, nYear0 As Long, rYear0 As Double, nYear1 As Long, rYear1 As Double, rYearElse As Double) As Double

    Application.Volatile
    Dim in0 As Long, in1 As Long, in2 As Long
    Dim i As Long, n As Long, crg As Long
    Dim nTN As Long
    Dim TNCT() As Double
    
    On Error Resume Next
    
    crg = Cashflow.Count

    ReDim TNCT(1 To crg)

    For i = 1 To crg
        TNCT(i) = Cashflow.Cells(1, i).Value
    Next i

    For n = 1 To crg
        If TNCT(n) > 0 Then
            nTN = n
            in0 = nTN + nYear0
            in1 = in0 + nYear1
            in2 = crg
            Exit For
        End If
    Next n

    Select Case nTax
    Case 1 To nTN - 1
        ThueTN = 0
    Case nTN To in0 - 1
        ThueTN = TNCT(nTax) * rYear0
    Case in0 To in1 - 1
        ThueTN = TNCT(nTax) * rYear1
    Case in1 To in2
        ThueTN = TNCT(nTax) * rYearElse
    End Select
    
End Function
